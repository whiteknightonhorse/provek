    # Dispatcher operational script (GL-05, Fable GL-00 ruling-2 §2 "live check"): deploy + 3 live writes on Bradbury.
    # Reads the key from key.hex; never prints it. Outside the repo on purpose.
    import hashlib, json, time, urllib.request
    from eth_account import Account
    from genlayer_py import create_client
    from genlayer_py.chains import testnet_bradbury
    from gltest.types import CalldataAddress, TransactionStatus
    from gltest.utils import extract_contract_address
    import os
    # genlayer-py 0.18.0 (latest on PyPI, 2026-09-28) lacks status 14 = LEADER_REVEALING, which Bradbury emits
    # (genlayer-js src/types/transactions.ts has it). Non-terminal; treat as REVEALING while polling only.
    import genlayer_py.types.transactions as _gtx
    _gtx.TRANSACTION_STATUS_NUMBER_TO_NAME.setdefault('14', _gtx.TransactionStatus.REVEALING)

    REPO = "/home/incubator/incubator"
    COMMIT = "5e28d5f933329c124c058996da28b2abd4da5d37"
    CONTRACT = f"{REPO}/integrations/genlayer-witness/contracts/provek_evidence_witness.py"
    LICENSE_URL = f"https://raw.githubusercontent.com/whiteknightonhorse/provek/{COMMIT}/LICENSE"
    SUBJECT = "git:whiteknightonhorse/provek"


    # Bradbury gen_call now returns {"data": <hex>, "status": ..., ...}; genlayer-py 0.18.0 expects a bare hex string.
    import eth_utils
    from genlayer_py.abi import calldata as _cd
    from genlayer_py.abi.transactions import serialize as _ser
    from genlayer_py.contracts.actions import make_calldata_object as _mco
    def gl_read(client, addr, fn, args=None):
        data = _ser([_cd.encode(_mco(method=fn, args=args, kwargs=None)), b"\x00"])
        res = client.provider.make_request(method="gen_call", params=[{"type": "read", "to": addr,
              "from": client.local_account.address, "data": data, "transaction_hash_variant": "latest-nonfinal"}])["result"]
        hexdata = res["data"] if isinstance(res, dict) else res
        return _cd.decode(eth_utils.hexadecimal.decode_hex("0x" + hexdata))

    def log(*a): print(time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()), *a, flush=True)

    acct = Account.from_key(open("/home/incubator/orchestra/gl/key.hex").read().strip())
    log("deployer", acct.address, "chain", testnet_bradbury.id)
    client = create_client(chain=testnet_bradbury, account=acct)

    lic = urllib.request.urlopen(LICENSE_URL, timeout=30).read()
    lic_sha = hashlib.sha256(lic).hexdigest()
    log("LICENSE", LICENSE_URL, "bytes", len(lic), "sha256", lic_sha)

    if os.environ.get("CONTRACT_ADDR"):
        addr = os.environ["CONTRACT_ADDR"]
        log("using already-deployed contract", addr, "(deploy tx", os.environ.get("DEPLOY_TX"), ")")
    else:
        tx = client.deploy_contract(code=open(CONTRACT).read(), args=[CalldataAddress(acct.address)])
        log("deploy tx", tx)
        rc = client.wait_for_transaction_receipt(tx, status=TransactionStatus.ACCEPTED, interval=5000, retries=180)
        # testnet: tx_data_decoded is None, gltest's extract_contract_address fails; deploy tx recipient = new contract
        addr = rc.get("recipient")
        log("DEPLOYED contract", addr, "status", rc.get("status_name"))
    log("supported_criteria (liveness)", gl_read(client, addr, "supported_criteria"))

    writes = [
      ("PASS expected: artifact_hash of pinned LICENSE", "artifact_hash", LICENSE_URL, lic_sha),
      ("PASS expected: url_reachable provek.dev",         "url_reachable", "https://provek.dev/", ""),
      ("FAIL expected: artifact_hash with a wrong digest", "artifact_hash", LICENSE_URL, "0"*64),
    ]
    for label, crit, url, exp in writes:
        t = client.write_contract(address=addr, function_name="witness", args=[SUBJECT, crit, url, exp])
        log("write", label, "tx", t)
        r = client.wait_for_transaction_receipt(t, status=TransactionStatus.ACCEPTED, interval=5000, retries=180)
        log("  status", r.get("status_name"))

    ids = json.loads(gl_read(client, addr, "list_by_subject", [SUBJECT]))
    log("list_by_subject", ids)
    for i in ids:
        log("get_result", i, gl_read(client, addr, "get_result", [i]))
    log("supported_criteria", gl_read(client, addr, "supported_criteria"))
    log("DONE contract", addr)

(`LEADER_REVEALING`), which `genlayer-py==0.18.0`'s own `TRANSACTION_STATUS_NUMBER_TO_NAME` map does

never taught, not a failure of the deploy itself. The deploy transaction that this attempt submitted
is the same one this record uses (§5) - it was already accepted on-chain by the time attempt 3 ran;
only the polling code in this script needed the fix in §1 (the `setdefault('14', ...)` patch present
in the script quoted above, added after this attempt).
