import { chromium } from 'playwright-core';
const browser = await chromium.launch({ executablePath: '/usr/bin/chromium-browser', headless: true, args: ['--no-sandbox'] });
const page = await browser.newPage();
await page.goto('https://provek.dev/', { waitUntil: 'networkidle' });

for (const w of [360, 768, 1440]) {
  await page.setViewportSize({ width: w, height: 900 });
  await page.waitForTimeout(200);
  const data = await page.evaluate(() => {
    const grid = document.querySelector('main .grid.gap-10');
    if (!grid) return { error: 'grid not found' };
    const cols = Array.from(grid.children);
    const rects = cols.map(c => c.getBoundingClientRect());
    return { gridRect: grid.getBoundingClientRect(), colHeights: rects.map(r => r.height) };
  });
  console.log('WIDTH', w, JSON.stringify(data, null, 2));
}
await browser.close();

import { chromium } from 'playwright-core';
const browser = await chromium.launch({ executablePath: '/usr/bin/chromium-browser', headless: true, args: ['--no-sandbox'] });
const page = await browser.newPage();
await page.goto('https://provek.dev/', { waitUntil: 'networkidle' });
await page.setViewportSize({ width: 1440, height: 900 });
await page.waitForTimeout(300);
const data = await page.evaluate(() => {
  const grid = document.querySelector('main .grid.gap-10');
  const [leftSection, aside] = grid.children;
  const leftLastChild = leftSection.lastElementChild;
  const asideLastVisible = aside.querySelector('p:last-of-type') || aside.lastElementChild;
  const stepSection = Array.from(document.querySelectorAll('main section')).find(s => s.textContent.includes('Step 1'));
  return {
    gridBottom: grid.getBoundingClientRect().bottom,
    leftContentBottom: leftLastChild.getBoundingClientRect().bottom,
    asideContentBottom: asideLastVisible.getBoundingClientRect().bottom,
    stepSectionTop: stepSection.getBoundingClientRect().top,
  };
});
console.log(JSON.stringify(data, null, 2));
console.log('empty gap under left column (px):', data.stepSectionTop - data.leftContentBottom);
await browser.close();

  await page.waitForTimeout(200);
  const data = await page.evaluate(() => {
    const grid = document.querySelector('main .grid.gap-10');
    if (!grid) return { error: 'grid not found' };
    const cols = Array.from(grid.children);
    const rects = cols.map(c => c.getBoundingClientRect());
    const buildSection = Array.from(document.querySelectorAll('main section')).find(s => s.textContent.includes('What you can build today'));
    const whySection = Array.from(document.querySelectorAll('main section')).find(s => s.textContent.includes('Why this is worth your time today'));
    return {
      gridRect: grid.getBoundingClientRect(),
      colHeights: rects.map(r => r.height),
      buildSectionBottom: buildSection ? buildSection.getBoundingClientRect().bottom : null,
      whySectionTop: whySection ? whySection.getBoundingClientRect().top : null,
    };
  });
  console.log('WIDTH', w, JSON.stringify(data, null, 2));
}
await browser.close();

CONCLUSION: at >=1024px, the left column's box height is still stretched to match the right
column (CSS Grid's `align-items: stretch` default, unchanged, not this task's target) - but the
left column's own VISIBLE CONTENT now reaches that same height (1082.7 = 1082.7 at 1440px), where
before it stopped 390px short. The remaining 56px before "Why this is worth your time today" is
the section's own `mt-14` margin, present at every width, not a defect. At 360/768 the columns
still stack (unchanged, <1024px never exhibited the gap): the left column is simply taller now
because it carries more content (the templates section), which is the intended effect of D3, not
a regression.
